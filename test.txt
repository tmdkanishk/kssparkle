var paymentMap, paymentMarkers = [];
var defaultLat = 30.9, defaultLng = 75.8;
var paymentMapLoaded = false;

function paymentMapInit(lat, lng) {
  var canvas = document.getElementById('payment-map-canvas');
  if (!canvas) return;

  var latNum = parseFloat(lat), lngNum = parseFloat(lng);
  if (isNaN(latNum) || isNaN(lngNum)) { latNum = defaultLat; lngNum = defaultLng; }

  var initialPoint = new google.maps.LatLng(latNum, lngNum);
  var mapOptions = { zoom: 15, center: initialPoint };

  paymentMap = new google.maps.Map(canvas, mapOptions);

  google.maps.event.addListener(paymentMap, 'click', function(event) {
    paymentAddMarker(event.latLng);
    paymentSaveData(event);
  });

  paymentAddMarker(initialPoint);

  var searchInput = document.getElementById('input-payment-address-1');
  if (searchInput) {
    var autocomplete = new google.maps.places.Autocomplete(searchInput, { types: ['geocode'] });
    google.maps.event.addListener(autocomplete, 'place_changed', function() {
      var place = autocomplete.getPlace();
      if (place.geometry && place.geometry.location) {
        paymentMap.setCenter(place.geometry.location);
        paymentAddMarker(place.geometry.location);
        $('#input-latitude-payment').val(place.geometry.location.lat());
        $('#input-longitude-payment').val(place.geometry.location.lng());
      }
    });
  }
}

function paymentAddMarker(location) {
  paymentMarkers.forEach(function(m) { m.setMap(null); });
  paymentMarkers = [];
  var marker = new google.maps.Marker({ position: location, map: paymentMap, draggable: true });
  google.maps.event.addListener(marker, 'dragend', function(event) { paymentSaveData(event); });
  paymentMarkers.push(marker);
}

function paymentSaveData(event) {
  var lat = event.latLng.lat(), lng = event.latLng.lng();
  $('#input-latitude-payment').val(lat);
  $('#input-longitude-payment').val(lng);

  var geocoder = new google.maps.Geocoder();
  geocoder.geocode({ location: { lat: lat, lng: lng } }, function(results, status) {
    if (status === 'OK' && results[0]) {
      $('#input-payment-address-1').val(results[0].formatted_address);
    }
  });
}

/* 🔥 init ONLY when map div becomes visible */
function initpaymentMapWhenVisible() {
  if (paymentMapLoaded) return;
  var el = document.getElementById('payment-map-canvas');
  if (!el) return;

  var rect = el.getBoundingClientRect();
  if (rect.width > 0 && rect.height > 0) {
    paymentMapLoaded = true;
    var whLat = '{{ latitude }}', whLng = '{{ longitude }}';
    if (whLat && whLng && !isNaN(parseFloat(whLat))) {
      paymentMapInit(whLat, whLng);
    } else {
      paymentMapInit(defaultLat, defaultLng);
    }
  }
}

$(window).on('load', function() {
  setTimeout(initpaymentMapWhenVisible, 800);
});

$(document).on('click change', function() {
  setTimeout(initpaymentMapWhenVisible, 500);
});

              backgroundColor: !isLiquidGlassSupported ? 'rgba(255,255,255,0.08)' : 'transparent',

               
              },
            ]}
          >
            {title && <Text style={styles.title}>{title}</Text>}
            <View style={[styles.content, style]}>{children}</View>

          </View>
        </LiquidGlassView>

    </View>
  );
};
